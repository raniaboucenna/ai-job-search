% --- Entry macros: a fixed-width date box anchors every entry to the same left
% column without a real multi-column/table layout, so pdftotext extraction
% stays in natural top-to-bottom, left-to-right reading order. ---
\newcommand{\datebox}[1]{\makebox[3.3cm][l]{\small\color{accent}\textbf{#1}}}

% #1 date range (ASCII hyphen, e.g. 2024-Present) #2 role title #3 organization #4 location
% NOTE: the \\[-8pt] pulls the following \begin{itemize} tight against the org/location
% line - keep it only when an itemize block immediately follows (see eduentry below, which
% has no list after it and uses a small positive skip instead).
\newcommand{\cventry}[4]{%
\noindent\datebox{#1}\textbf{#2}\\
\hspace*{3.3cm}{\small\color{subtext}\textit{#3} -- #4}\\[-8pt]}

% #1 date range #2 degree #3 institution, location
\newcommand{\eduentry}[3]{%
\noindent\datebox{#1}\textbf{#2}\\
\hspace*{3.3cm}{\small\color{subtext}#3}\\[3pt]}
